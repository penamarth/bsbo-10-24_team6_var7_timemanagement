View -> VM : SelectedEmployee (setter)

activate Facade

Facade -> Service : GetWorkloadReport(employeeId, weekStart)
activate Service

Service -> Storage : GetTasksByAssignee(employeeId)
activate Storage
Storage --> Service : List<TaskItem>
deactivate Storage

alt Данные успешно получены
    Service -> DTO : Создать WorkloadRow(...)
    activate DTO

    deactivate Service

    Facade --> VM : Exception
